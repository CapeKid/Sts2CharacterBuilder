    private static async Task Dialogue()
    {
#if BUILDER_AUTHORED_DIALOGUE
        var own = ModelDb.Character<Character>().Id;
        var foreign = ModelDb.Character<MegaCrit.Sts2.Core.Models.Characters.Ironclad>().Id;
        // A native result is compared by reference with our postfix disabled only
        // for this isolated test. No production fallback algorithm is duplicated.
        var method = AccessTools.Method(typeof(AncientDialogueSet), nameof(AncientDialogueSet.GetValidDialogues));
        var patch = AccessTools.Method(typeof(CharacterDialogue), "Select");
        var owner = Harmony.GetPatchInfo(method)?.Postfixes.SingleOrDefault(p => p.PatchMethod == patch)?.owner
            ?? throw new InvalidDataException("Authored dialogue hook missing; inspect the mod initializer log.");
        var harmony = new Harmony(owner);
        foreach (var ancient in ModelDb.AllAncients) {
            var set = ancient.DialogueSet;
            foreach (var character in new[] { own, foreign }) foreach (var visits in new[] { 0, 1, 100 }) foreach (var allow in new[] { false, true }) {
                var authored = character == own && Request.GetProperty("dialogue").EnumerateArray().Any(r =>
                    r.GetProperty("ancient").GetString() == ancient.Id.Entry && r.GetProperty("visit").GetString() == (visits == 0 ? "first" : "repeat"));
                var actual = set.GetValidDialogues(character, visits, 200, allow).ToArray();
                if (authored) { Require(actual.Length == 1 && actual[0].Lines.All(l => l.LineText!.LocEntryKey.Contains(".BUILDER_DIALOGUE.")), "Authored selector missed."); continue; }
                harmony.Unpatch(method, patch);
                AncientDialogue[] expected;
                try { expected = set.GetValidDialogues(character, visits, 200, allow).ToArray(); }
                finally { harmony.Patch(method, postfix: new HarmonyMethod(patch)); }
                Require(actual.SequenceEqual(expected), "Native dialogue fallback changed.");
            }
        }
        var unknown = new AncientDialogueSet { FirstVisitEverDialogue = null, CharacterDialogues = new(), AgnosticDialogues = [] };
        Require(!unknown.GetValidDialogues(own, 0, 0, false).Any(), "Untagged empty set changed.");
        var loc = MegaCrit.Sts2.Core.Localization.LocManager.Instance;
        var language = loc.Language;
        var other = MegaCrit.Sts2.Core.Localization.LocManager.Languages.First(l => l != language);
        try {
            loc.SetLanguage(other);
            CheckInlineLocalization();
            foreach (var row in Request.GetProperty("dialogue").EnumerateArray().Where(r => r.GetProperty("ancient").GetString() != "THE_ARCHITECT")) {
                var ancient = ModelDb.AllAncients.Single(a => a.Id.Entry == row.GetProperty("ancient").GetString());
                var visits = row.GetProperty("visit").GetString() == "first" ? 0 : 100;
                var selected = ancient.DialogueSet.GetValidDialogues(own, visits, 0, false).Single();
                CheckDialogueModel(row, selected.Lines, other);
            }
        } finally { loc.SetLanguage(language); }
        CheckInlineLocalization();
        Checks.Add(new { check = "dialogue-controls", foreignUnchanged = true, omittedUnchanged = true, untaggedUnchanged = true,
            lateRepeat = true, firstEver = true, localeSwitch = other, inlineLocalizationPreserved = true });
        var observer = new Harmony(typeof(ModEntry).Namespace + ".dialogue-sfx-observer");
        observer.Patch(AccessTools.Method(typeof(NAncientDialogueLine), nameof(NAncientDialogueLine.PlaySfx)),
            postfix: new HarmonyMethod(typeof(NativeProbe), nameof(CountDialogueSfx)));
        try {
            foreach (var row in Request.GetProperty("dialogue").EnumerateArray().Where(r => r.GetProperty("ancient").GetString() != "THE_ARCHITECT"))
                foreach (var locale in DialogueLocales(row)) {
                    loc.SetLanguage(locale); CheckInlineLocalization();
                    await DialogueEncounter(row, locale);
                }
        } finally { loc.SetLanguage(language); observer.UnpatchAll(observer.Id); }
        await EndingDialogues();
#else
        await Task.CompletedTask;
        Checks.Add(new { check = "dialogue-controls", skipped = true, reason = "No authored dialogue" });
#endif
    }

    private sealed record EndingSelection(AncientDialogue Native, ArchitectAttackers Start, ArchitectAttackers End, int? Visit, bool Repeating);
    private static readonly System.Runtime.CompilerServices.ConditionalWeakTable<MegaCrit.Sts2.Core.Models.Events.TheArchitect, EndingSelection> EndingSelections = new();
    private static bool ForceEndingRoom;
    private static bool EndingRoomRoll(MegaCrit.Sts2.Core.Map.MapPointType __0, ref RoomType __result)
    {
        if (!ForceEndingRoom || __0 != MegaCrit.Sts2.Core.Map.MapPointType.Unknown) return true;
        __result = RoomType.Event; return false;
    }
    private static void ObserveEnding(MegaCrit.Sts2.Core.Models.Events.TheArchitect __instance, AncientDialogue ____dialogue)
    {
        EndingSelections.Remove(__instance);
        EndingSelections.Add(__instance, new(____dialogue, ____dialogue.StartAttackers, ____dialogue.EndAttackers, ____dialogue.VisitIndex, ____dialogue.IsRepeating));
    }
    private static AncientDialogue EndingLines(MegaCrit.Sts2.Core.Models.Events.TheArchitect model) =>
        (AncientDialogue)AccessTools.Property(model.GetType(), "Dialogue").GetValue(model)!;
    private static int EndingIndex(MegaCrit.Sts2.Core.Models.Events.TheArchitect model) =>
        (int)AccessTools.Property(model.GetType(), "CurrentLineIndex").GetValue(model)!;
    private static bool EndingExpected(Player player, int wins) => player.Character is Character
        && Request.GetProperty("dialogue").EnumerateArray().Any(r => r.GetProperty("ancient").GetString() == "THE_ARCHITECT"
            && r.GetProperty("visit").GetString() == (wins == 0 ? "first" : "repeat"));
    private static void CheckEndingSelection(MegaCrit.Sts2.Core.Models.Events.TheArchitect model, bool authored)
    {
        Require(EndingSelections.TryGetValue(model, out var before), "Native ending selection was not observed.");
        var after = EndingLines(model);
        Require(after.StartAttackers == before!.Start && after.EndAttackers == before.End
            && after.VisitIndex == before.Visit && after.IsRepeating == before.Repeating, "Ending choreography/visit metadata changed.");
        Require(authored ? after.Lines.All(l => l.LineText!.LocEntryKey.Contains(".BUILDER_DIALOGUE.THE_ARCHITECT."))
            : ReferenceEquals(after, before.Native), "Ending selection/fallback changed.");
    }
    private static async Task EndingDialogues()
    {
        var rows = Request.GetProperty("dialogue").EnumerateArray().Where(r => r.GetProperty("ancient").GetString() == "THE_ARCHITECT").ToArray();
        // Even ordinary-only configurations check that our selector leaves endings alone.
        var observer = new Harmony(typeof(ModEntry).Namespace + ".ending-observer");
        observer.Patch(AccessTools.Method(typeof(MegaCrit.Sts2.Core.Models.Events.TheArchitect), "LoadDialogue"),
            postfix: new HarmonyMethod(typeof(NativeProbe), nameof(ObserveEnding)) { priority = Priority.First });
        observer.Patch(AccessTools.Method(typeof(RunManager), "RollRoomTypeFor"),
            prefix: new HarmonyMethod(typeof(NativeProbe), nameof(EndingRoomRoll)));
        var progress = SaveManager.Instance.Progress;
        var ownStats = progress.GetOrCreateCharacterStats(ModelDb.Character<Character>().Id); var priorWins = ownStats.TotalWins;
        try {
            foreach (var character in new CharacterModel[] { ModelDb.Character<Character>(), ModelDb.Character<MegaCrit.Sts2.Core.Models.Characters.Ironclad>() }) {
                var player = Player.CreateForNewRun(character, Run.Players.Single().UnlockState, 104);
                _ = RunState.CreateForTest([player], ModelDb.ActsByIndex.Select(a => a.First()).ToArray(), [], GameMode.Standard, 0, "BUILDER-ENDING-CONTROL");
                var stats = progress.GetOrCreateCharacterStats(character.Id); var original = stats.TotalWins;
                try {
                    foreach (var wins in new[] { 0, 1, 100 }) {
                        stats.TotalWins = wins;
                        var model = (MegaCrit.Sts2.Core.Models.Events.TheArchitect)ModelDb.Event<MegaCrit.Sts2.Core.Models.Events.TheArchitect>().ToMutable();
                        await model.BeginEvent(player, null!, false);
                        CheckEndingSelection(model, EndingExpected(player, wins));
                    }
                } finally { stats.TotalWins = original; }
            }
            foreach (var row in rows) {
                var visit = row.GetProperty("visit").GetString()!;
                Progress("ending:" + visit);
                await NGame.Instance!.ReturnToMainMenu();
                ownStats.TotalWins = visit == "first" ? 0 : 100;
                await NGame.Instance.StartNewSingleplayerRun(ModelDb.Character<Character>(), true,
                    ModelDb.ActsByIndex.Select(a => a.First()).ToArray(), [], "BUILDER-ENDING-" + visit, GameMode.Standard, 0);
                Run.ExtraFields.StartedWithNeow = true;
                var rooms = (RoomSet)AccessTools.Field(typeof(ActModel), "_rooms").GetValue(Run.Act)!;
                AccessTools.Field(rooms.GetType(), "events").SetValue(rooms, new List<EventModel> { ModelDb.Event<MegaCrit.Sts2.Core.Models.Events.TheArchitect>() });
                AccessTools.Field(rooms.GetType(), "eventsVisited").SetValue(rooms, 0);
                var point = Run.Map.StartingMapPoint; point.PointType = MegaCrit.Sts2.Core.Map.MapPointType.Unknown;
                ForceEndingRoom = true;
                try { await RunManager.Instance.EnterMapCoord(point.coord); }
                finally { ForceEndingRoom = false; }
                var model = await EndingReady(row);
                // Reload from a real pre-entry map checkpoint after partial advancement.
                if (row.GetProperty("lines").GetArrayLength() > 1) await AdvanceEnding(model, 1);
                await NGame.Instance.ReturnToMainMenu();
                var saved = SaveManager.Instance.LoadRunSave();
                Require(saved.Success && saved.SaveData != null, "Ending checkpoint missing.");
                var restored = RunState.FromSerializable(saved.SaveData!);
                await RunManager.Instance.SetUpSavedSingleplayer(restored, saved.SaveData!);
                await NGame.Instance.LoadRun(restored, saved.SaveData!.PreFinishedRoom);
                model = await EndingReady(row);
                Require(EndingIndex(model) == 0, "Ending checkpoint did not replay from start.");
                for (int i = 1; i < row.GetProperty("lines").GetArrayLength(); i++) {
                    await AdvanceEnding(model, i); CheckEndingText(row, model);
                }
                var option = model.CurrentOptions.Single();
                var native = (MegaCrit.Sts2.Core.Events.EventOption)AccessTools.Method(model.GetType(), "CreateProceedOption").Invoke(model, null)!;
                Require(option.IsProceed && !option.IsLocked && option.TextKey == native.TextKey
                    && option.Title.LocEntryKey == native.Title.LocEntryKey && option.OnChosen == native.OnChosen,
                    "Native Proceed/WinRun callback changed.");
                Checks.Add(new { check = "ending-dialogue", ancient = "THE_ARCHITECT", visit, visibleText = true, speakersPreserved = true,
                    nativeAdvance = true, nativeChoreography = true, proceedUnchanged = true, checkpointReplay = true });
            }
            Checks.Add(new { check = "ending-controls", foreignUnchanged = true, omittedUnchanged = true, lateRepeat = true, nativeMetadata = true });
        } finally { ownStats.TotalWins = priorWins; ForceEndingRoom = false; observer.UnpatchAll(observer.Id); }
    }
    private static async Task<MegaCrit.Sts2.Core.Models.Events.TheArchitect> EndingReady(JsonElement row)
    {
        await Wait(() => Run.CurrentRoom is EventRoom r && r.LocalMutableEvent is MegaCrit.Sts2.Core.Models.Events.TheArchitect m
            && m.CurrentOptions.Count == 1 && DialogueNodes(Tree.Root).OfType<NEventOptionButton>().Any(b => b.IsVisibleInTree() && b.IsEnabled && b.Option == m.CurrentOptions[0]), "Ending button missing.");
        var model = (MegaCrit.Sts2.Core.Models.Events.TheArchitect)((EventRoom)Run.CurrentRoom!).LocalMutableEvent;
        CheckEndingSelection(model, true); CheckEndingText(row, model);
        return model;
    }
    private static void CheckEndingText(JsonElement row, MegaCrit.Sts2.Core.Models.Events.TheArchitect model)
    {
        CheckDialogueModel(row, EndingLines(model).Lines, "eng");
        var i = EndingIndex(model);
        var bubble = (MegaCrit.Sts2.Core.Nodes.Vfx.NSpeechBubbleVfx)AccessTools.Property(model.GetType(), "SpeechBubble").GetValue(model)!;
        Require(bubble != null && bubble.IsInsideTree(), "Ending speech bubble missing.");
        var label = (MegaCrit.Sts2.addons.mega_text.MegaRichTextLabel)AccessTools.Field(bubble.GetType(), "_label").GetValue(bubble)!;
        Require(label.Text == row.GetProperty("lines")[i].GetProperty("text").GetString(), "Rendered ending speech differs.");
        if (i < row.GetProperty("lines").GetArrayLength() - 1)
            Require(model.CurrentOptions.Single().Title.GetFormattedText() == DialogueText(row, "eng", i, "next"), "Ending next label differs.");
    }
    private static async Task AdvanceEnding(MegaCrit.Sts2.Core.Models.Events.TheArchitect model, int index)
    {
        var option = model.CurrentOptions.Single();
        var button = DialogueNodes(Tree.Root).OfType<NEventOptionButton>().Single(b => b.IsVisibleInTree() && b.Option == option);
        Require(button.IsEnabled && !option.IsProceed, "Ending advance unavailable.");
        await MegaCrit.Sts2.Core.AutoSlay.Helpers.UiHelper.Click(button);
        await RunManager.Instance.EventSynchronizer.AwaitPendingOptionTasks();
        await Wait(() => EndingIndex(model) == index && model.CurrentOptions.Count == 1 && model.CurrentOptions[0] != option
            && DialogueNodes(Tree.Root).OfType<NEventOptionButton>().Any(b => b.IsVisibleInTree() && b.IsEnabled && b.Option == model.CurrentOptions[0]), "Ending advance failed.");
    }
    private static int DialogueSfxCalls;
    private static void CountDialogueSfx() => DialogueSfxCalls++;
    private static void CheckLiveDialogueLocale(JsonElement row, NAncientEventLayout layout, string language)
    {
        if (!row.TryGetProperty("translations", out _)) return;
        var loc = MegaCrit.Sts2.Core.Localization.LocManager.Instance;
        var other = DialogueLocales(row).First(l => l != language);
        var index = DialogueIndex(layout); var currentLines = DialogueLines(layout).ToArray(); var sfx = DialogueSfxCalls;
        try {
            loc.SetLanguage(other); CheckInlineLocalization(); CheckRenderedDialogue(row, layout, other);
            Require(DialogueIndex(layout) == index && DialogueLines(layout).SequenceEqual(currentLines), "Locale switch replayed or replaced the conversation.");
        } finally { loc.SetLanguage(language); }
        CheckRenderedDialogue(row, layout, language);
        Require(DialogueIndex(layout) == index && DialogueSfxCalls == sfx, "Locale switch advanced dialogue or replayed sound.");
    }
    private static string[] DialogueLocales(JsonElement row)
    {
        if (!row.TryGetProperty("translations", out var variants)) return ["eng"];
        var languages = variants.EnumerateArray().Select(t => t.GetProperty("language").GetString()!).Append("eng").ToHashSet(StringComparer.Ordinal);
        Require(languages.All(l => MegaCrit.Sts2.Core.Localization.LocManager.Languages.Contains(l)), "Authored locale unavailable on this game build.");
        var fallback = MegaCrit.Sts2.Core.Localization.LocManager.Languages.OrderBy(l => l, StringComparer.Ordinal).FirstOrDefault(l => !languages.Contains(l));
        if (fallback != null) languages.Add(fallback);
        return languages.OrderBy(l => l, StringComparer.Ordinal).ToArray();
    }
    private static string DialogueText(JsonElement row, string language, int index, string field)
    {
        if (row.TryGetProperty("translations", out var variants))
            foreach (var variant in variants.EnumerateArray())
                if (variant.GetProperty("language").GetString() == language && variant.GetProperty("lines")[index].TryGetProperty(field, out var translated))
                    return translated.GetString()!;
        return row.GetProperty("lines")[index].TryGetProperty(field, out var english) ? english.GetString()! : "Continue";
    }
    private static void CheckDialogueModel(JsonElement row, IReadOnlyList<AncientDialogueLine> lines, string language)
    {
        var expected = row.GetProperty("lines");
        Require(lines.Count == expected.GetArrayLength(), "Translated dialogue line count differs.");
        for (int i = 0; i < lines.Count; i++) {
            Require(lines[i].LineText!.GetFormattedText() == DialogueText(row, language, i, "text")
                && lines[i].Speaker.ToString() == expected[i].GetProperty("speaker").GetString(), "Dialogue text/speaker differs for " + language);
            if (i < lines.Count - 1) Require(lines[i].NextButtonText!.GetFormattedText() == DialogueText(row, language, i, "next"), "Translated next label differs.");
        }
    }
    private static void CheckRenderedDialogue(JsonElement row, NAncientEventLayout layout, string language)
    {
        CheckDialogueModel(row, DialogueLines(layout), language);
        var container = (VBoxContainer)AccessTools.Field(typeof(NAncientEventLayout), "_dialogueContainer").GetValue(layout)!;
        var nodes = container.GetChildren().OfType<NAncientDialogueLine>().ToArray();
        Require(nodes.Length == row.GetProperty("lines").GetArrayLength(), "Rendered dialogue node count differs.");
        for (int i = 0; i < nodes.Length; i++)
            Require(nodes[i].GetNode<MegaCrit.Sts2.addons.mega_text.MegaRichTextLabel>("%Text").Text == DialogueText(row, language, i, "text"), "Rendered dialogue label differs for " + language);
        var index = DialogueIndex(layout);
        if (index < nodes.Length - 1) {
            var label = (MegaCrit.Sts2.addons.mega_text.MegaLabel)AccessTools.Field(typeof(NAncientEventLayout), "_fakeNextButtonLabel").GetValue(layout)!;
            Require(label.Text == DialogueText(row, language, index, "next"), "Rendered next label differs for " + language);
        }
    }
    private static void CheckInlineLocalization()
    {
        Require(new MegaCrit.Sts2.Core.Localization.LocString("characters", ModelDb.Character<Character>().Id.Entry + ".title").Exists(), "Character title lost after locale reload.");
        foreach (var row in Request.GetProperty("cards").EnumerateArray()) {
            var type = typeof(ModEntry).Assembly.GetType(typeof(ModEntry).Namespace + ".Card_" + row.GetProperty("id").GetString(), true)!;
            var key = ModelDb.GetId(type).Entry;
            Require(new MegaCrit.Sts2.Core.Localization.LocString("cards", key + ".title").GetFormattedText() == row.GetProperty("name").GetString(), "Card title lost after locale reload.");
            Require(new MegaCrit.Sts2.Core.Localization.LocString("cards", key + ".description").Exists(), "Card description lost after locale reload.");
        }
        foreach (var (group, table, prefix) in new[] { ("relics", "relics", "Relic_"), ("potions", "potions", "Potion_") })
            foreach (var row in Request.GetProperty("items").GetProperty(group).EnumerateArray()) {
                var type = typeof(ModEntry).Assembly.GetType(typeof(ModEntry).Namespace + "." + prefix + row.GetProperty("id").GetString(), true)!;
                Require(new MegaCrit.Sts2.Core.Localization.LocString(table, ModelDb.GetId(type).Entry + ".title").GetFormattedText() == row.GetProperty("name").GetString(), "Item title lost after locale reload.");
            }
    }
    private static IEnumerable<Node> DialogueNodes(Node node)
    {
        yield return node;
        foreach (var child in node.GetChildren()) foreach (var descendant in DialogueNodes(child)) yield return descendant;
    }
    private static NAncientEventLayout? DialogueLayout() => DialogueNodes(Tree.Root).OfType<NAncientEventLayout>().SingleOrDefault(l => l.IsVisibleInTree());
    private static List<AncientDialogueLine> DialogueLines(NAncientEventLayout layout) =>
        (List<AncientDialogueLine>)AccessTools.Field(typeof(NAncientEventLayout), "_dialogue").GetValue(layout)!;
    private static int DialogueIndex(NAncientEventLayout layout) => (int)AccessTools.Field(typeof(NAncientEventLayout), "_currentDialogueLine").GetValue(layout)!;
    private static async Task CheckDialogue(JsonElement row, bool advance, string language)
    {
        var rows = row.GetProperty("lines").EnumerateArray().ToArray();
        await Wait(() => DialogueLayout() is {} l && DialogueLines(l).Count == rows.Length, "Ancient layout dialogue missing.");
        var layout = DialogueLayout()!; var lines = DialogueLines(layout);
        CheckRenderedDialogue(row, layout, language);
        CheckLiveDialogueLocale(row, layout, language);
        if (!advance) return;
        Require(DialogueIndex(layout) == 0, "Dialogue did not start on first line.");
        for (var i = 1; i < rows.Length; i++) {
            var hitbox = (NAncientDialogueHitbox)AccessTools.Field(typeof(NAncientEventLayout), "_dialogueHitbox").GetValue(layout)!;
            await Wait(() => hitbox.IsVisibleInTree() && hitbox.IsEnabled, "Dialogue advance unavailable.");
            Require(!string.IsNullOrEmpty(hitbox.GetHotkey()), "Native dialogue hotkey missing.");
            hitbox.EmitSignal(NClickableControl.SignalName.Released, hitbox);
            var index = i;
            await Wait(() => DialogueIndex(layout) == index, "Dialogue advance failed.");
            CheckRenderedDialogue(row, layout, language);
            CheckLiveDialogueLocale(row, layout, language);
        }
        await Wait(() => layout.OptionButtons.Any(b => b.IsVisibleInTree() && b.IsEnabled), "Ancient reward options unavailable after dialogue.");
    }
    private static async Task DialogueEncounter(JsonElement row, string language)
    {
        var entry = row.GetProperty("ancient").GetString()!; var visit = row.GetProperty("visit").GetString()!;
        Progress("dialogue:" + entry + "/" + visit + "/" + language);
        await NGame.Instance!.ReturnToMainMenu();
        var ancient = ModelDb.AllAncients.Single(a => a.Id.Entry == entry);
        var stats = SaveManager.Instance.Progress.GetOrCreateAncientStats(ancient.Id);
        stats.CharStats.Clear();
        stats.CharStats.Add(new AncientCharacterStats { Character = ModelDb.Character<Character>().Id, Losses = visit == "first" ? 0 : 100 });
        await NGame.Instance.StartNewSingleplayerRun(ModelDb.Character<Character>(), true,
            ModelDb.ActsByIndex.Select(a => a.First()).ToArray(), [], "BUILDER-DIALOGUE-" + entry + visit + language, GameMode.Standard, 0);
        Run.ExtraFields.StartedWithNeow = true;
        var rooms = (RoomSet)AccessTools.Field(typeof(ActModel), "_rooms").GetValue(Run.Act)!;
        rooms.Ancient = ancient;
        var point = Run.Map.StartingMapPoint; point.PointType = MegaCrit.Sts2.Core.Map.MapPointType.Ancient;
        var rewards = ancient.AllPossibleOptions.Select(o => o.TextKey).ToArray();
        await RunManager.Instance.EnterMapCoord(point.coord);
        await CheckDialogue(row, true, language);
        Require(rewards.SequenceEqual(ancient.AllPossibleOptions.Select(o => o.TextKey)), "Ancient reward catalog changed.");
        if (language == "eng") Checks.Add(new { check = "authored-dialogue", ancient = entry, visit, visibleText = true, nativeAdvance = true, rewardsAvailable = true });
        if (row.TryGetProperty("translations", out _)) Checks.Add(new { check = "dialogue-locale", ancient = entry, visit, language,
            visibleText = true, renderedLabels = true, nextLabels = true, speakersPreserved = true, liveSwitch = true, progressPreserved = true, noSoundReplay = true,
            nativeAdvance = true, rewardsAvailable = true, inlineLocalizationPreserved = true });
        await NGame.Instance.ReturnToMainMenu();
        var saved = SaveManager.Instance.LoadRunSave();
        Require(saved.Success && saved.SaveData != null, "Ancient checkpoint missing.");
        var restored = RunState.FromSerializable(saved.SaveData!);
        await RunManager.Instance.SetUpSavedSingleplayer(restored, saved.SaveData!);
        await NGame.Instance.LoadRun(restored, saved.SaveData!.PreFinishedRoom);
        await CheckDialogue(row, true, language);
        if (language == "eng") Checks.Add(new { check = "dialogue-resume", ancient = entry, visit, nativeCheckpoint = true, replayFromStart = true });
        if (row.TryGetProperty("translations", out _)) Checks.Add(new { check = "dialogue-locale-resume", ancient = entry, visit, language,
            nativeCheckpoint = true, replayFromStart = true, renderedLabels = true, nativeAdvance = true, rewardsAvailable = true });
    }
