    private static bool ForceEventRoll;
    private static bool EventRoomRoll(MegaCrit.Sts2.Core.Map.MapPointType __0, ref RoomType __result)
    {
        if (!ForceEventRoll || __0 != MegaCrit.Sts2.Core.Map.MapPointType.Unknown) return true;
        __result = RoomType.Event;
        return false;
    }

    private static async Task Events()
    {
#if BUILDER_AUTHORED_EVENTS
        var player = Run.Players.Single();
        var acts = ModelDb.ActsByIndex.Select(a => a.First()).ToArray();
        var foreign = RunState.CreateForTest([Player.CreateForNewRun(ModelDb.Character<MegaCrit.Sts2.Core.Models.Characters.Ironclad>(), player.UnlockState, 101)],
            acts, [], GameMode.Standard, 0, "BUILDER-EVENT-FOREIGN");
        var coop = RunState.CreateForTest([Player.CreateForNewRun(ModelDb.Character<Character>(), player.UnlockState, 102),
            Player.CreateForNewRun(ModelDb.Character<Character>(), player.UnlockState, 103)], acts, [], GameMode.Standard, 0, "BUILDER-EVENT-COOP");
        foreach (var row in Request.GetProperty("events").EnumerateArray()) {
            var canonical = Event(row);
            Require(canonical.IsAllowed(Run) && !canonical.IsAllowed(foreign) && !canonical.IsAllowed(coop), "Event character/solo eligibility mismatch.");
            foreach (var (variants, index) in ModelDb.ActsByIndex.Select((variants, index) => (variants, index)))
                foreach (var act in variants) Require(act.AllEvents.Any(e => e.Id == canonical.Id)
                    == row.GetProperty("acts").EnumerateArray().Any(a => a.GetInt32() == index + 1), "Event act registration mismatch.");
            foreach (var (page, data) in EventPages(row)) {
                foreach (var choice in data.GetProperty("choices").EnumerateArray()) {
                    Progress("event:" + row.GetProperty("id").GetString() + "/" + page + "/" + choice.GetProperty("id").GetString());
                    if (choice.TryGetProperty("requires", out _)) await ConditionalEventChoice(row, page, choice);
                    else await EventChoice(row, page, choice);
                }
                await EventLeave(row, page);
                await EventResume(row, page);
            }
        }
        Checks.Add(new { check = "event-pools", characterOnly = true, soloOnly = true, actsVerified = true });
#else
        await Task.CompletedTask;
        Checks.Add(new { check = "event-pools", skipped = true, reason = "No authored events" });
#endif
    }

#if BUILDER_AUTHORED_EVENTS
    private static EventModel Event(JsonElement row) => ModelDb.GetById<EventModel>(ModelDb.GetId(
        typeof(ModEntry).Assembly.GetType(typeof(ModEntry).Namespace + ".Event_" + row.GetProperty("id").GetString(), true)!));
    private static IEnumerable<Godot.Node> EventNodes(Godot.Node node)
    {
        yield return node;
        foreach (var child in node.GetChildren()) foreach (var descendant in EventNodes(child)) yield return descendant;
    }
    private static MegaCrit.Sts2.Core.Nodes.Events.NEventOptionButton EventButton(MegaCrit.Sts2.Core.Events.EventOption option) =>
        EventNodes(Tree.Root).OfType<MegaCrit.Sts2.Core.Nodes.Events.NEventOptionButton>()
            .Single(b => b.IsVisibleInTree() && b.Option == option);
    private static MegaCrit.Sts2.Core.Events.EventOption EventOption(EventModel model, string choice) =>
        model.CurrentOptions.Single(o => o.TextKey.EndsWith(".options." + choice, StringComparison.Ordinal));
    private static async Task<EventModel> EventRoom(JsonElement row)
    {
        await RunManager.Instance.EnterRoomDebug(RoomType.Event, model: Event(row), showTransition: false);
        return await EventReady();
    }
    private static async Task<EventModel> EventReady()
    {
        await Wait(() => Run.CurrentRoom is MegaCrit.Sts2.Core.Rooms.EventRoom room && room.LocalMutableEvent?.CurrentOptions.Count > 0,
            "Authored event options did not arrive.");
        var model = ((MegaCrit.Sts2.Core.Rooms.EventRoom)Run.CurrentRoom!).LocalMutableEvent;
        await Wait(() => EventNodes(Tree.Root).OfType<MegaCrit.Sts2.Core.Nodes.Events.NEventOptionButton>()
            .Count(b => b.IsVisibleInTree() && model.CurrentOptions.Contains(b.Option)) == model.CurrentOptions.Count, "Event buttons missing.");
        Require(model.Title.Exists() && model.InitialDescription.Exists() && !string.IsNullOrWhiteSpace(model.InitialDescription.GetFormattedText()), "Event localization missing.");
        var portrait = model.CreateInitialPortrait();
        Require(portrait != null && portrait.GetWidth() > 0, "Native event fallback portrait missing.");
        return model;
    }
    private static async Task EventStats(int gold, int hp, int maximum = 200)
    {
        var player = Run.Players.Single();
        player.Creature.SetMaxHpInternal(maximum); player.Creature.SetCurrentHpInternal(hp);
        await PlayerCmd.SetGold(gold, player);
    }
    private static (decimal gold, decimal hp) EventStats() => (Run.Players.Single().Gold, Run.Players.Single().Creature.CurrentHp);
    private static IEnumerable<(string Id, JsonElement Data)> EventPages(JsonElement row) =>
        new[] { ("INITIAL", row) }.Concat(row.TryGetProperty("pages", out var pages)
            ? pages.EnumerateArray().Select(p => (p.GetProperty("id").GetString()!, p)) : []);
    private static async Task ClickEvent(EventModel model, string choice, string? next = null)
    {
        var option = EventOption(model, choice);
        await Wait(() => EventNodes(Tree.Root).OfType<MegaCrit.Sts2.Core.Nodes.Events.NEventOptionButton>()
            .Any(b => b.IsVisibleInTree() && b.Option == option), "Current event button missing.");
        Require(!option.IsLocked && EventButton(option).IsEnabled, "Affordable event button disabled.");
        await MegaCrit.Sts2.Core.AutoSlay.Helpers.UiHelper.Click(EventButton(option));
        await Wait(() => next == null ? model.IsFinished : !model.IsFinished && model.CurrentOptions.Count > 0
            && model.CurrentOptions.All(o => o.TextKey.StartsWith($"{model.Id.Entry}.pages.{next}.options.", StringComparison.Ordinal)),
            "Event choice did not reach its destination.");
        await RunManager.Instance.EventSynchronizer.AwaitPendingOptionTasks();
    }
    private static async Task NavigateEvent(EventModel model, JsonElement row, string target)
    {
        var paths = new Dictionary<string, List<JsonElement>> { ["INITIAL"] = [] };
        var queue = new Queue<string>(); queue.Enqueue("INITIAL");
        while (queue.Count > 0) {
            var page = queue.Dequeue();
            foreach (var choice in EventPages(row).Single(p => p.Id == page).Data.GetProperty("choices").EnumerateArray()) {
                if (!choice.TryGetProperty("next", out var next) || paths.ContainsKey(next.GetString()!)) continue;
                paths[next.GetString()!] = [.. paths[page], choice]; queue.Enqueue(next.GetString()!);
            }
        }
        var current = "INITIAL";
        foreach (var choice in paths[target]) {
            await SatisfyEvent(choice); RefreshEventPage(model, current);
            await ClickEvent(model, choice.GetProperty("id").GetString()!, choice.GetProperty("next").GetString());
            current = choice.GetProperty("next").GetString()!;
        }
        Require(model.Description.GetFormattedText() == EventPages(row).Single(p => p.Id == target).Data.GetProperty("body").GetString(),
            "Branch page text differs.");
    }
    private static async Task<EventModel> EventPage(JsonElement row, string page)
    {
        var before = EventStats(); var maximum = Run.Players.Single().Creature.MaxHp;
        var model = await EventRoom(row);
        await NavigateEvent(model, row, page);
        await EventStats((int)before.gold, (int)before.hp, (int)maximum); RefreshEventPage(model, page);
        return model;
    }
    private static async Task FinishEvent(EventModel model, JsonElement row, string page)
    {
        for (int step = 0; step < 8; step++) {
            var choice = EventPages(row).Single(p => p.Id == page).Data.GetProperty("choices")[0];
            var next = choice.TryGetProperty("next", out var value) ? value.GetString() : null;
            await SatisfyEvent(choice); RefreshEventPage(model, page);
            await ClickEvent(model, choice.GetProperty("id").GetString()!, next);
            if (next == null) return;
            page = next;
        }
        throw new InvalidDataException("Event path exceeded its page bound.");
    }
    private static void RefreshEventPage(EventModel model, string page) =>
        AccessTools.Method(typeof(AuthoredEvent), "ShowPage").Invoke(model, [page]);
    private static (int gold, int hp) PositiveEventStats(JsonElement choice)
    {
        var gold = choice.GetProperty("cost").GetProperty("gold").GetInt32();
        var hp = Math.Max(100, choice.GetProperty("cost").GetProperty("hp").GetInt32() + 1);
        if (choice.TryGetProperty("requires", out var conditions)) {
            if (conditions.TryGetProperty("min_gold", out var minimum)) gold = Math.Max(gold, minimum.GetInt32());
            if (conditions.TryGetProperty("min_hp_percent", out var health)) hp = Math.Max(hp, health.GetInt32() * 100);
        }
        return (gold, hp);
    }
    private static async Task SatisfyEvent(JsonElement choice)
    {
        if (!choice.TryGetProperty("requires", out _)) return;
        var stats = PositiveEventStats(choice);
        await EventStats(stats.gold, stats.hp, 10000);
    }
    private static async Task ConditionalEventChoice(JsonElement row, string page, JsonElement choice)
    {
        var id = choice.GetProperty("id").GetString()!; var positive = PositiveEventStats(choice);
        var conditions = choice.GetProperty("requires").EnumerateObject().ToArray();
        var cases = conditions.Select(c => (name: c.Name, value: c.Value.GetInt32())).ToList();
        foreach (var key in new[] { "gold", "hp" })
            if (choice.GetProperty("cost").GetProperty(key).GetInt32() > 0)
                cases.Add(("cost_" + key, choice.GetProperty("cost").GetProperty(key).GetInt32()));
        foreach (var control in cases) {
            await EventStats(positive.gold, positive.hp, 10000); var model = await EventPage(row, page);
            var gold = positive.gold; var hp = positive.hp;
            switch (control.name) {
                case "min_gold": case "cost_gold": gold = control.value - 1; break;
                case "max_gold": gold = control.value + 1; break;
                case "min_hp_percent": hp = control.value * 100 - 1; break;
                case "max_hp_percent": hp = control.value * 100 + 1; break;
                case "cost_hp": hp = control.value; break;
            }
            var old = EventOption(model, id);
            await EventStats(gold, hp, 10000); var before = EventStats();
            await old.Chosen();
            Require(EventStats() == before && !model.IsFinished && EventOption(model, id).IsLocked, "Changed conditions charged or navigated.");
            await LockedEvent(model, id);
            var options = model.CurrentOptions.ToArray();
            await EventStats(positive.gold, positive.hp, 10000); before = EventStats();
            await old.Chosen();
            Require(EventStats() == before && !model.IsFinished && model.CurrentOptions.SequenceEqual(options), "Old conditional callback revived after refresh.");
            // Rebuild at the inclusive boundary, with all other requirements satisfied.
            if (control.name == "max_gold") await EventStats(control.value, positive.hp, 10000);
            if (control.name == "max_hp_percent") await EventStats(positive.gold, control.value * 100, 10000);
            RefreshEventPage(model, page);
            Require(!EventOption(model, id).IsLocked, "Inclusive condition boundary was locked.");
            if (!control.name.StartsWith("cost_", StringComparison.Ordinal))
                Checks.Add(new { check = "event-condition", id = row.GetProperty("id").GetString(), page, choice = id, condition = control.name,
                    inclusive = true, outsideLocked = true, staleGuard = true, oldCallbackRejected = true });
        }
        await EventStats(positive.gold, positive.hp, 10000);
        await EventOutcome(row, page, choice, await EventPage(row, page));
    }
    private static async Task EventChoice(JsonElement row, string page, JsonElement choice)
    {
        var id = choice.GetProperty("id").GetString()!;
        var next = choice.TryGetProperty("next", out var value) ? value.GetString() : null;
        var gold = choice.GetProperty("cost").GetProperty("gold").GetInt32();
        var hp = choice.GetProperty("cost").GetProperty("hp").GetInt32();
        if (gold > 0) {
            await EventStats(gold - 1, 100); var locked = await EventPage(row, page);
            await LockedEvent(locked, id);
        }
        if (hp > 0) {
            await EventStats(gold, hp); var locked = await EventPage(row, page);
            await LockedEvent(locked, id);
        }
        if (gold > 0 || hp > 0) {
            await EventStats(gold, 100); var stale = await EventPage(row, page);
            var option = EventOption(stale, id);
            await EventStats(gold > 0 ? gold - 1 : gold, gold > 0 ? 100 : hp);
            var before = EventStats();
            await option.Chosen();
            Require(EventStats() == before && !stale.IsFinished && EventOption(stale, id).IsLocked, "Stale affordability check charged resources.");
            var refreshed = stale.CurrentOptions.ToArray();
            await EventStats(gold, 100); before = EventStats();
            await option.Chosen();
            Require(EventStats() == before && !stale.IsFinished && stale.CurrentOptions.SequenceEqual(refreshed), "Old option executed after an affordability refresh.");
        }
        await EventStats(gold, 100);
        var model = await EventPage(row, page);
        await EventOutcome(row, page, choice, model);
    }
    private static async Task EventOutcome(JsonElement row, string page, JsonElement choice, EventModel model)
    {
        var id = choice.GetProperty("id").GetString()!;
        var next = choice.TryGetProperty("next", out var value) ? value.GetString() : null;
        var gold = choice.GetProperty("cost").GetProperty("gold").GetInt32();
        var hp = choice.GetProperty("cost").GetProperty("hp").GetInt32();
        var selected = EventOption(model, id); var siblings = model.CurrentOptions.Where(o => !o.IsLocked).ToArray();
        var initial = EventStats();
        Require(selected.Title.Exists() && selected.Description.Exists(), "Event choice localization missing.");
        await ClickEvent(model, id, next);
        var expectedGold = initial.gold - gold + choice.GetProperty("reward").GetProperty("gold").GetInt32();
        var expectedHp = Math.Min(Run.Players.Single().Creature.MaxHp, initial.hp - hp + choice.GetProperty("reward").GetProperty("heal").GetInt32());
        Require(EventStats() == (expectedGold, expectedHp), "Event cost/reward differs.");
        var text = next == null ? choice.GetProperty("result").GetString()
            : EventPages(row).Single(p => p.Id == next).Data.GetProperty("body").GetString();
        Require(model.Description.GetFormattedText() == text, "Event result/page text differs.");
        var options = model.CurrentOptions.ToArray();
        await Task.WhenAll(siblings.Select(option => option.Chosen()));
        Require(EventStats() == (expectedGold, expectedHp) && model.Description.GetFormattedText() == text
            && model.CurrentOptions.SequenceEqual(options) && model.IsFinished == (next == null), "Old/sibling options replayed or changed the event.");
        Checks.Add(new { check = "authored-event", id = row.GetProperty("id").GetString(), page, choice = id, next,
            thresholds = true, staleGuard = true, nativeButton = true, exactEffects = true, duplicateSuppressed = true, pageText = true });
    }
    private static async Task LockedEvent(EventModel model, string choice)
    {
        var option = EventOption(model, choice); var before = EventStats();
        await Wait(() => EventNodes(Tree.Root).OfType<MegaCrit.Sts2.Core.Nodes.Events.NEventOptionButton>().Any(b => b.IsVisibleInTree() && b.Option == option), "Locked event button missing.");
        Require(option.IsLocked, "Unaffordable event choice was not locked: " + choice + " / " + before);
        // Native locked options remain enabled for hover text; OnRelease rejects them.
        await MegaCrit.Sts2.Core.AutoSlay.Helpers.UiHelper.Click(EventButton(option));
        await Tree.ToSignal(Tree, SceneTree.SignalName.ProcessFrame);
        Require(!option.WasChosen && !model.IsFinished && EventStats() == before, "Locked event choice executed or charged resources.");
    }
    private static async Task EventLeave(JsonElement row, string page)
    {
        await EventStats(0, 1); var model = await EventPage(row, page); var before = EventStats();
        await ClickEvent(model, "Leave");
        Require(EventStats() == before, "Leave changed event resources.");
        Checks.Add(new { check = "event-leave", id = row.GetProperty("id").GetString(), page, alwaysAvailable = true, unchanged = true });
    }
    private static async Task ReloadEventCheckpoint()
    {
        await NGame.Instance!.ReturnToMainMenu();
        var save = SaveManager.Instance.LoadRunSave();
        Require(save.Success && save.SaveData != null, "Event checkpoint unreadable.");
        var restored = RunState.FromSerializable(save.SaveData!);
        await RunManager.Instance.SetUpSavedSingleplayer(restored, save.SaveData!);
        await NGame.Instance!.LoadRun(restored, save.SaveData!.PreFinishedRoom);
    }
    private static async Task EventResume(JsonElement row, string page)
    {
        // Each checkpoint control gets its own map/run, regardless of library size.
        await NGame.Instance!.ReturnToMainMenu();
        await NGame.Instance.StartNewSingleplayerRun(ModelDb.Character<Character>(), true,
            ModelDb.ActsByIndex.Select(a => a.First()).ToArray(), [], "BUILDER-EVENT-CHECKPOINT-" + row.GetProperty("id").GetString() + "-" + page, GameMode.Standard, 0);
        Run.ExtraFields.StartedWithNeow = true;
        await EventStats(300, 100);
        // Debug rooms do not update the saved map position. Select a known event
        // in this disposable room set and enter through the native map pipeline.
        var rooms = (MegaCrit.Sts2.Core.Rooms.RoomSet)AccessTools.Field(typeof(ActModel), "_rooms").GetValue(Run.Act)!;
        AccessTools.Field(rooms.GetType(), "events").SetValue(rooms, new List<EventModel> { Event(row) });
        AccessTools.Field(rooms.GetType(), "eventsVisited").SetValue(rooms, 0);
        ForceEventRoll = true;
        try {
            var point = Run.Map.StartingMapPoint;
            point.PointType = MegaCrit.Sts2.Core.Map.MapPointType.Unknown;
            Progress("event-checkpoint:" + row.GetProperty("id").GetString() + ":enter");
            await RunManager.Instance.EnterMapCoord(point.coord);
            await EventReady();
            var checkpoint = EventStats(); var checkpointMaxHp = Run.Players.Single().Creature.MaxHp;
            await ReloadEventCheckpoint();
            var model = await EventReady();
            Require(EventStats() == checkpoint && Run.Players.Single().Creature.MaxHp == checkpointMaxHp, "Initial event checkpoint changed resources: " + checkpoint + " / " + EventStats());
            Require(model.Id == Event(row).Id && model.Description.GetFormattedText() == row.GetProperty("body").GetString(), "Resumed event identity/initial page differs.");
            await NavigateEvent(model, row, page);
            // A branch is transient: reloading before its terminal outcome returns to INITIAL.
            await ReloadEventCheckpoint(); model = await EventReady();
            Require(EventStats() == checkpoint && Run.Players.Single().Creature.MaxHp == checkpointMaxHp && model.Description.GetFormattedText() == row.GetProperty("body").GetString(), "Branch checkpoint did not restore INITIAL/resources.");
            await NavigateEvent(model, row, page);
            await FinishEvent(model, row, page);
            var outcome = EventStats(); var outcomeMaxHp = Run.Players.Single().Creature.MaxHp;
            // Interrupt before leaving; the native checkpoint must restore the pre-choice state.
            await ReloadEventCheckpoint();
            model = await EventReady();
            Require(EventStats() == checkpoint && Run.Players.Single().Creature.MaxHp == checkpointMaxHp && model.Description.GetFormattedText() == row.GetProperty("body").GetString(), "Interrupted visit kept a branch or uncommitted rewards.");
            await NavigateEvent(model, row, page);
            await FinishEvent(model, row, page);
            Require(EventStats() == outcome, "Replayed checkpoint duplicated event rewards.");
            var next = Run.CurrentMapPoint!.Children.First();
            next.PointType = MegaCrit.Sts2.Core.Map.MapPointType.RestSite;
            await RunManager.Instance.EnterMapCoord(next.coord);
            await ReloadEventCheckpoint();
            await Wait(() => Run.CurrentRoom != null, "Saved next room did not arrive.");
            Require(EventStats() == outcome && Run.Players.Single().Creature.MaxHp == outcomeMaxHp && Run.CurrentRoom is not MegaCrit.Sts2.Core.Rooms.EventRoom, "Completed visit did not persist after leaving.");
            Checks.Add(new { check = "event-resume", id = row.GetProperty("id").GetString(), page, branchRollback = true, initialCheckpoint = true,
                interruptedRollback = true, replayNoDuplicate = true, completedPersistence = true });
        } finally { ForceEventRoll = false; }
    }
