internal static class CharacterDialogue
{
    private sealed record Identity(string Entry);
    private static readonly ConditionalWeakTable<AncientDialogueSet, Identity> Sets = new();
    private static readonly JsonElement Rows = JsonDocument.Parse(@DIALOGUE_JSON@).RootElement.Clone();
    private static bool Initialized;
    internal static void Initialize()
    {
        if (Initialized) return;
        var harmony = new Harmony(@MOD_ID@ + ".character-dialogue");
        try {
            harmony.Patch(AccessTools.PropertyGetter(typeof(AncientEventModel), nameof(AncientEventModel.DialogueSet)),
                postfix: new HarmonyMethod(typeof(CharacterDialogue), nameof(Identify)));
            harmony.Patch(AccessTools.Method(typeof(AncientDialogueSet), nameof(AncientDialogueSet.GetValidDialogues)),
                postfix: new HarmonyMethod(typeof(CharacterDialogue), nameof(Select)));
            harmony.Patch(AccessTools.Method(typeof(LocManager), "TriggerLocaleChange"),
                prefix: new HarmonyMethod(typeof(CharacterDialogue), nameof(RestoreLocalization)),
                postfix: new HarmonyMethod(typeof(CharacterDialogue), nameof(RefreshVisibleDialogue)));
            Initialized = true;
        } catch { harmony.UnpatchAll(harmony.Id); throw; }
    }
    private static void Identify(AncientEventModel __instance, AncientDialogueSet __result) =>
        Sets.GetValue(__result, _ => new Identity(__instance.Id.Entry));

    private static string Prefix => @MOD_ID@ + ".BUILDER_DIALOGUE.";
    private static string Key(JsonElement row, int index) => Prefix + row.GetProperty("ancient").GetString() + "." + row.GetProperty("visit").GetString() + "." + index;
    private static void Install(JsonElement row)
    {
        var lines = row.GetProperty("lines").EnumerateArray().ToArray();
        JsonElement[]? localized = null;
        if (row.TryGetProperty("translations", out var variants))
            foreach (var variant in variants.EnumerateArray())
                if (variant.GetProperty("language").GetString() == LocManager.Instance.Language)
                    localized = variant.GetProperty("lines").EnumerateArray().ToArray();
        var entries = new Dictionary<string, string>();
        for (var i = 0; i < lines.Length; i++) {
            var key = Key(row, i);
            entries[key] = localized != null && localized[i].TryGetProperty("text", out var text)
                ? text.GetString()! : lines[i].GetProperty("text").GetString()!;
            if (i < lines.Length - 1)
                entries[key + ".next"] = localized != null && localized[i].TryGetProperty("next", out var label) ? label.GetString()!
                    : lines[i].TryGetProperty("next", out var english) ? english.GetString()! : "Continue";
        }
        LocManager.Instance.GetTable("ancients").MergeWith(entries);
    }
    private static void RestoreLocalization()
    {
        // Restore owned keys before locale-change listeners observe fresh tables.
        foreach (var row in Rows.EnumerateArray()) Install(row);
    }
    private static IEnumerable<Node> Nodes(Node root)
    {
        yield return root;
        foreach (var child in root.GetChildren()) foreach (var node in Nodes(child)) yield return node;
    }
    private static bool Owned(AncientDialogueLine line) => line.LineText?.LocEntryKey.StartsWith(Prefix, StringComparison.Ordinal) == true;
    private static void RefreshVisibleDialogue()
    {
        if (Engine.GetMainLoop() is not SceneTree tree) return;
        var nodes = Nodes(tree.Root).ToArray();
        foreach (var node in nodes.OfType<NAncientDialogueLine>().Where(n => n.IsInsideTree())) {
            var line = (AncientDialogueLine)AccessTools.Field(typeof(NAncientDialogueLine), "_line").GetValue(node)!;
            if (Owned(line)) node.GetNode<MegaRichTextLabel>("%Text").Text = line.LineText!.GetFormattedText();
        }
        foreach (var layout in nodes.OfType<NAncientEventLayout>().Where(n => n.IsInsideTree())) {
            var lines = (List<AncientDialogueLine>)AccessTools.Field(typeof(NAncientEventLayout), "_dialogue").GetValue(layout)!;
            if (lines.Count > 0 && lines.All(Owned))
                AccessTools.Method(typeof(NAncientEventLayout), "UpdateFakeNextButton").Invoke(layout, null);
        }
        // Text/label refresh only: no replay, advance, SetDialogue or sound calls.
    }
    private static void Select(AncientDialogueSet __instance, ModelId characterId, int charVisits,
        ref IEnumerable<AncientDialogue> __result)
    {
        if (characterId != ModelDb.Character<Character>().Id || !Sets.TryGetValue(__instance, out var identity)) return;
        var visit = charVisits <= 0 ? "first" : "repeat";
        foreach (var row in Rows.EnumerateArray()) {
            if (row.GetProperty("ancient").GetString() != identity.Entry || row.GetProperty("visit").GetString() != visit) continue;
            var lines = row.GetProperty("lines").EnumerateArray().ToArray();
            var dialogue = new AncientDialogue(lines.Select(_ => "").ToArray()) { VisitIndex = charVisits <= 0 ? 0 : 1, IsRepeating = charVisits > 0 };
            Install(row);
            for (var i = 0; i < lines.Length; i++) {
                var key = Key(row, i);
                var line = dialogue.Lines[i];
                line.Speaker = Enum.Parse<AncientDialogueSpeaker>(lines[i].GetProperty("speaker").GetString()!);
                line.LineText = new LocString("ancients", key);
                if (i < lines.Length - 1) {
                    line.NextButtonText = new LocString("ancients", key + ".next");
                }
            }
            __result = [dialogue];
            return;
        }
    }
}
